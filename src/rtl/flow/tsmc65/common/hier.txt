# Block table of the P8b hierarchical unit flow (syn/dc_unit.tcl, syn/run_hier.sh).  One block per line:

# units (syn/run_hier.sh <unit>): fp gemm vec lowa mem rng ctrl; chip is the P8c top (prepared, not run in P8b).
