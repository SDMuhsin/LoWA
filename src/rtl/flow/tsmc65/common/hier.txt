# Block table of the P8b hierarchical unit flow (syn/dc_unit.tcl, syn/run_hier.sh).  One block per line:
#   name  module  params  children  io_pct
# module: an RTL module, or `@<block>/<instance>` = the sub-design of that instance in <block>'s elaboration.
# params: elaboration parameters (`-` = module defaults), written so that the elaborated design name equals the
#   one the parent's instantiation produces (full16 values from top_chip / docs/calc/budget.py cfg full16).
# children: blocks compiled before this one and linked into it as mapped, dont_touch sub-designs (each unique
#   sub-block is compiled once and reused by every parent); `-` = none (the block is compiled flat).
# io_pct: input / output delay on the block's own ports in % of the period (sdc/gen_sdc.tcl).  Units use 20;
#   bank leaves use 40 because their address comes through the parent's control logic (tighter, not looser).
# link (optional 6th column): `abs` = children are linked as block abstractions (interface logic only); every
#   parent uses it.  Linking full netlists with dont_touch was tried first: DC topographical then places the
#   children's cells spread over the parent's whole area and, the children being dont_touch, cannot re-buffer
#   the longer estimated wires (vu16 -2.49 ns and mm -1.97 ns inside lanes / mask generators that close
#   alone).  With abstractions each child's internal paths are timed in its own run (a compact region, as a
#   hierarchical layout places it) and the parent times the interface paths; the unit totals add each child's
#   own summary.  It also keeps gm16 within memory.
# units (syn/run_hier.sh <unit>): fp gemm vec lowa mem rng ctrl; chip is the P8c top (prepared, not run in P8b).
#
# ---- fp: the arithmetic units, standalone, at the configurations the design instantiates
fma      fp_fma        STAGES=>2                                 -                       20
add      fp_add        STAGES=>2                                 -                       20
mul2     fp_mul        STAGES=>2                                 -                       20
mul1     fp_mul        STAGES=>1                                 -                       20
cmp      fp_cmp        STAGES=>1                                 -                       20
i2f      fp_i2f        STAGES=>1                                 -                       20
sfu      fp_sfu        -                                         -                       20
# ---- latch register-file banks (tsmc65 backend), one per shape used in full16
bank256  mm_sram_bank  WORDS=>256,WIDTH=>32,AW=>8                -                       40
bank512  mm_sram_bank  WORDS=>512,WIDTH=>32,AW=>9                -                       40
bank768  mm_sram_bank  WORDS=>768,WIDTH=>32,AW=>10               -                       40
# ---- gemm: 16 x 16 PEs, NSFU 16
pe       gm_pe         FMA_STAGES=>2,LAMBDA=>4,QW=>2             -                       20

# for one run: each sub-unit is a block of its own, taken from gm16's elaboration (`@gm16/<instance>`)
gmarr    @gm16/u_array -                                         pe                      20 abs

gmep     @gm16/u_epilogue -                                      -                       20
gm16     gm_top        ROWS=>16,COLS=>16,NSFU=>16                gmarr,gmxf,gmwf,gmdr,gmep 20 abs
# ---- vec: V = 16 lanes
lane     vu_lane       NS=>8,SW=>3,RB=>256,RBW=>8,NOB=>2,LF=>2   bank256                 20 abs
vu16     vu_top        -                                         lane,bank256            20 abs
# ---- lowa
lw       lw_top        -                                         bank256,bank512,bank768 20 abs

rng9     rn_mkgen      MKW=>16,NCORE=>9,STAGES=>3                -                       20

mm       mm_top        NR=>8,NM=>2,NW=>4,AXL=>3,MK_DRAM=>0       rng9,rng3               20 abs
# ---- ctrl: sequencer (with ct_dma) and the host CSR slave
